## 1. Load template and create presentation

```python
import json
from pathlib import Path
from pptx import Presentation
from pptx.util import Inches, Pt, Emu
from pptx.dml.color import RGBColor
from pptx.enum.text import PP_ALIGN

TEMPLATE_PATH = Path("refs/slide_template/template.json")
with open(TEMPLATE_PATH) as f:
    tmpl = json.load(f)

def hex_to_rgb(hex_str):
    h = hex_str.lstrip("#")
    return RGBColor(int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16))

# Standard widescreen (13.33" x 7.5")
prs = Presentation()
prs.slide_width  = Inches(13.33)
prs.slide_height = Inches(7.5)
```

## 2. Helper: set text run style

```python
def style_run(run, font_face, size_pt, color_hex, bold=False):
    run.font.name = font_face
    run.font.size = Pt(size_pt)
    run.font.color.rgb = hex_to_rgb(color_hex)
    run.font.bold = bold

def fill_shape(shape, color_hex):
    shape.fill.solid()
    shape.fill.fore_color.rgb = hex_to_rgb(color_hex)
```

## 3. Title slide

```python
def add_title_slide(prs, title_text, subtitle_text):
    layout = prs.slide_layouts[6]  # blank layout
    slide = prs.slides.add_slide(layout)

    W, H = prs.slide_width, prs.slide_height
    bg_color = tmpl["colors"]["background_title"]

    # Full-bleed background
    bg = slide.shapes.add_shape(1, 0, 0, W, H)  # 1 = rectangle
    fill_shape(bg, bg_color)
    bg.line.fill.background()  # no border

    # Title text box
    txb = slide.shapes.add_textbox(Inches(1), Inches(2.5), Inches(10), Inches(1.5))
    tf = txb.text_frame
    tf.word_wrap = True
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = title_text
    style_run(run, tmpl["fonts"]["title_face"], tmpl["fonts"]["title_size_pt"],
               tmpl["colors"]["text_light"], bold=True)
    p.alignment = PP_ALIGN.LEFT

    # Subtitle
    txb2 = slide.shapes.add_textbox(Inches(1), Inches(4.2), Inches(10), Inches(1))
    tf2 = txb2.text_frame
    p2 = tf2.paragraphs[0]
    run2 = p2.add_run()
    run2.text = subtitle_text
    style_run(run2, tmpl["fonts"]["subtitle_face"], tmpl["fonts"]["subtitle_size_pt"],
               tmpl["colors"]["text_light"])

    return slide
```

## 5. Content (bullet) slide

```python
def add_content_slide(prs, title, bullets):
    """
    bullets: list of strings
    """
    layout = prs.slide_layouts[6]
    slide = prs.slides.add_slide(layout)
    W, H = prs.slide_width, prs.slide_height

    # White background
    bg = slide.shapes.add_shape(1, 0, 0, W, H)
    fill_shape(bg, tmpl["colors"]["background_content"])
    bg.line.fill.background()

    # Title bar
    bar = slide.shapes.add_shape(1, 0, 0, W, Inches(1.1))
    fill_shape(bar, tmpl["colors"]["primary"])
    bar.line.fill.background()

    txb_title = slide.shapes.add_textbox(Inches(0.4), Inches(0.15), Inches(12), Inches(0.8))
    tf = txb_title.text_frame
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = title
    style_run(run, tmpl["fonts"]["title_face"], 28, tmpl["colors"]["text_light"], bold=True)

    # Bullet body
    txb_body = slide.shapes.add_textbox(Inches(0.6), Inches(1.4), Inches(11.5), Inches(5.5))
    tf_body = txb_body.text_frame
    tf_body.word_wrap = True
    for i, bullet in enumerate(bullets):
        p = tf_body.paragraphs[0] if i == 0 else tf_body.add_paragraph()
        p.space_before = Pt(6)
        run = p.add_run()
        run.text = f"\u2022  {bullet}"
        style_run(run, tmpl["fonts"]["body_face"], tmpl["fonts"]["body_size_pt"],
                   tmpl["colors"]["text_dark"])

    return slide
```

## 6. Chart slide

```python
def add_chart_slide(prs, title, chart_png_path, caption=""):
    """
    chart_png_path: path to a PNG generated by plot_style.save()
    caption: one-line takeaway shown below the chart
    """
    layout = prs.slide_layouts[6]
    slide = prs.slides.add_slide(layout)
    W, H = prs.slide_width, prs.slide_height

    bg = slide.shapes.add_shape(1, 0, 0, W, H)
    fill_shape(bg, tmpl["colors"]["background_content"])
    bg.line.fill.background()

    bar = slide.shapes.add_shape(1, 0, 0, W, Inches(1.1))
    fill_shape(bar, tmpl["colors"]["primary"])
    bar.line.fill.background()

    txb_title = slide.shapes.add_textbox(Inches(0.4), Inches(0.15), Inches(12), Inches(0.8))
    tf = txb_title.text_frame
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = title
    style_run(run, tmpl["fonts"]["title_face"], 28, tmpl["colors"]["text_light"], bold=True)

    # Chart image — leave room for caption if provided
    img_bottom = Inches(6.8) if caption else Inches(7.3)
    slide.shapes.add_picture(str(chart_png_path),
                              Inches(0.5), Inches(1.25),
                              width=Inches(12.3),
                              height=img_bottom - Inches(1.25))

    if caption:
        txb_cap = slide.shapes.add_textbox(Inches(0.5), Inches(6.85), Inches(12.3), Inches(0.5))
        tf_cap = txb_cap.text_frame
        p_cap = tf_cap.paragraphs[0]
        run_cap = p_cap.add_run()
        run_cap.text = caption
        style_run(run_cap, tmpl["fonts"]["caption_face"], tmpl["fonts"]["caption_size_pt"],
                   tmpl["colors"]["text_dark"])
        p_cap.alignment = PP_ALIGN.CENTER

    return slide
```

## 7. Logo and footer (apply to all slides)

```python
from pptx.util import Inches
from PIL import Image  # for getting logo dimensions

def add_logo_and_footer(slide, prs, slide_number, total_slides, date_str):
    W, H = prs.slide_width, prs.slide_height
    logo_cfg = tmpl.get("logo", {})
    footer_cfg = tmpl.get("footer", {})

    # Logo
    logo_path = logo_cfg.get("path")
    if logo_path and Path(logo_path).exists():
        width_pct = logo_cfg.get("width_pct", 0.10)
        logo_w = W * width_pct
        img = Image.open(logo_path)
        aspect = img.height / img.width
        logo_h = logo_w * aspect

        pos = logo_cfg.get("position", "top_right")
        if pos == "top_right":
            left = W - logo_w - Inches(0.2)
            top  = Inches(0.1)
        elif pos == "bottom_right":
            left = W - logo_w - Inches(0.2)
            top  = H - logo_h - Inches(0.1)
        elif pos == "bottom_left":
            left = Inches(0.2)
            top  = H - logo_h - Inches(0.1)
        else:
            left, top = Inches(0.2), Inches(0.1)

        slide.shapes.add_picture(logo_path, left, top,
                                  width=int(logo_w), height=int(logo_h))

    # Footer text
    parts = []
    if footer_cfg.get("custom_text"):
        parts.append(footer_cfg["custom_text"])
    if footer_cfg.get("show_date"):
        parts.append(date_str)
    if footer_cfg.get("show_slide_number"):
        parts.append(f"{slide_number} / {total_slides}")

    if parts:
        footer_text = "   |   ".join(parts)
        txb = slide.shapes.add_textbox(Inches(0.3), H - Inches(0.35),
                                        W - Inches(0.6), Inches(0.3))
        tf = txb.text_frame
        p = tf.paragraphs[0]
        run = p.add_run()
        run.text = footer_text
        style_run(run, tmpl["fonts"]["caption_face"], 10,
                   tmpl["colors"].get("text_dark", "#333333"))
        p.alignment = PP_ALIGN.RIGHT
```
